36 // DocString: cmpYieldString::operator()
43 inline bool operator()(const std::string& p_first, const std::string& p_second) const noexcept
44 {
45 const size_t FIRST_LENGTH = p_first.length();
46 const size_t SECOND_LENGTH = p_second.length();
47 bool lessThan = false;
48 if (FIRST_LENGTH != SECOND_LENGTH)
49 {
50 lessThan = FIRST_LENGTH < SECOND_LENGTH;
51 }
52 else {
53 size_t i = FIRST_LENGTH;
54 bool gotValue = false;
55 while (!gotValue && i > 0)
56 {
57 --i;
58 if (p_first[i] != p_second[i])
59 {
60 if (p_first[i] < p_second[i])
61 {
62 lessThan = true;
63 }
64 else {
65 lessThan = false;
66 }
67 gotValue = true;
68 }
69 }
70 }
71 return lessThan;
72 }
73};
